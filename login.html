function getRoleFromEmail(email){ const e = email.toLowerCase(); const employerEmails = ['employer.jobway@gmail.com','admin@jobway.co.za','invoices@jobway.co.za']; if(employerEmails.includes(e)) return 'employer'; if(e.endsWith('@jobway.co.za')) return 'employer'; if(e.startsWith('employer.')) return 'employer'; return 'candidate'; } function login(){ let e=document.getElementById('email').value.trim().toLowerCase(); let pw=document.getElementById('pass').value; if(!e){alert('Enter email');return;} if(!pw){alert('Enter password');return;} let role = getRoleFromEmail(e); localStorage.removeItem('employer_email'); localStorage.removeItem('candidate_email'); localStorage.setItem('jobway_user_email', e); localStorage.setItem('jobway_logged_in','true'); localStorage.setItem('user_role', role); if(role === 'employer'){ localStorage.setItem('employer_email', e); location.href='employer-dashboard.html'; } else { localStorage.setItem('candidate_email', e); location.href='seeker-dashboard.html'; } }
